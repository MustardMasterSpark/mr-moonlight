#  press play; nothing has to be rebuilt and no C# has to be touched.
#
#  THE ONE RULE: every line fires and the director moves straight to the next.
#  The only line that ever blocks is `wait`. Scan the left column for it and you
#  know exactly where the level pauses.
#
#  Full verb reference: Docs/event-director.md

[main]

# A beat before the first message, so it does not land in the middle of the
# scene-load hitch.
wait  seconds=1

# Sets the objective AND announces it centre-bottom, in one line, so the words
# the player reads and the words the pause menu will hold can never drift apart.
# 20 is the real demo target (Carlos, 2026-09-02). 51 Spotters are placed on the
# island, so there is comfortable slack. If you change this number, change the
# wording in the same line — nothing keeps them in sync for you.
objective  kill_spotters  text="Homework: Kill 20 old timers"  kills=20  kind=Spotter  for=8

# Waits forever on purpose. This is a player-progress wait: it resolves when the
# twenty are dead and never otherwise, and timing it out would hand out a win
# nobody earned. See WaitVerb's class doc for the full timeout policy.
wait  objective=kill_spotters

win  "Good boy"
